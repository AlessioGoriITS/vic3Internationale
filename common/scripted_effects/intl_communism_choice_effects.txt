intl_clear_communism_law_visibility_vars = {
	if = { limit = { has_variable = bakunin_var } remove_variable = bakunin_var }
	if = { limit = { has_variable = proudhon_var } remove_variable = proudhon_var }
	if = { limit = { has_variable = bel_de_paepe_var } remove_variable = bel_de_paepe_var }
	if = { limit = { has_variable = blanqui_var } remove_variable = blanqui_var }
	if = { limit = { has_variable = karl_marx_var } remove_variable = karl_marx_var }
}

intl_create_or_update_internationale_movement = {
	if = {
		limit = {
			NOT = {
				any_political_movement = {
					is_political_movement_type = movement_internationale
				}
			}
		}
		create_political_movement = { type = movement_internationale }
	}
}

intl_set_anarchist_movement_line = {
	intl_create_or_update_internationale_movement = yes
	random_political_movement = {
		limit = {
			is_political_movement_type = movement_internationale
		}
		set_core_ideology = ideology_internationale_anarchist_movement
	}
}
