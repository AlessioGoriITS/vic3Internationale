je_internationale = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_global_international_situations

	widget = {
		gui = "gui/internationale_journal_button.gui"
		name = "internationale_journal_button"
		container = "custom_widget_container_1"
	}

	should_be_pinned_by_default_uninvolved_or_context = yes

	is_shown_when_inactive = {
	}

	possible = {
		any_country = {
			country_rank = rank_value:great_power
			has_law_or_variant = law_type:law_council_republic
		}
	}

	should_be_involved = {
		NOT = {
			is_involved_in_journal_entry = je_internationale
		}
		has_technology_researched = egalitarianism
		country_rank >= rank_value:major_power
	}

	immediate_all_involved = {
		set_variable = { name = intl_strength_anarchist value = 20 }
		set_variable = { name = intl_strength_mutualist value = 20 }
		set_variable = { name = intl_strength_syndicalist value = 20 }
		set_variable = { name = intl_strength_blanquist value = 20 }
		set_variable = { name = intl_strength_marxist value = 20 }

		# Base communist laws — one per lawgroup slot, permanently enabled from the start
		set_global_variable = { name = intl_law_marxist_council_republic value = 2 }
		set_global_variable = { name = intl_law_marxist_one_party_state value = 2 }
		set_global_variable = { name = intl_law_marxist_economy value = 2 }
		set_global_variable = { name = intl_law_marxist_bureaucracy value = 2 }
		set_global_variable = { name = intl_law_marxist_army value = 2 }

		hidden_effect = {
			intl_save_delegate_chars = yes
			intl_update_parliament = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			intl_save_delegate_chars = yes
			intl_update_parliament = yes
		}
	}

	complete = {
		custom_tooltip = {
			text = je_internationale_complete_tt
			always = no
		}
	}

	on_complete = {
	}

	weight = 1000
}


je_blanquism = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_internationalism

	widget = {
		gui = "gui/blanquist_journal_widgets.gui"
		name = "blanquist_preparation_widget"
		container = "custom_widget_container_1"
	}

	scripted_button = je_blanquism_prepare_cells
	scripted_button = je_blanquism_infiltrate_state
	scripted_button = je_blanquism_arm_cadres

	should_be_pinned_by_default_uninvolved_or_context = yes

	is_shown_when_inactive = {
	}

	immediate = {
		ig:ig_intelligentsia ?= {
			leader = {
				save_scope_as = blanquist_vanguard_leader
				set_ideology = ideology:ideology_blanquist
			}
		}
		set_variable = { name = blanquist_vanguard_leader_scope_gui value = scope:blanquist_vanguard_leader }
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_blanquism_status_desc
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = blanquist_cells_prepared_tt
			has_variable = blanquist_cells_prepared
		}
		custom_tooltip = {
			text = blanquist_state_infiltrated_tt
			has_variable = blanquist_state_infiltrated
		}
		custom_tooltip = {
			text = blanquist_cadres_armed_tt
			has_variable = blanquist_cadres_armed
		}
	}

	on_complete = {
		custom_tooltip = {
			text = blanquist_preparation_done_tt
			set_variable = blanquist_preparation_done
		}
		custom_tooltip = {
			text = blanquist_law_unlock_var_tt
			set_variable = blanqui_var
		}
		intl_unlock_blanquist_laws = yes
		add_modifier = {
			name = path_to_vanguardism
			days = normal_modifier_time
		}
	}

	weight = 1000
}
