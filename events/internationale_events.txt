namespace = mp_faction_events

# Fired only from je_mp_faction_leader (leader country). Other countries never receive this event.
mp_faction_events.1 = {
	type = country_event
	placement = root

	title = mp_faction_events.1.t
	desc = mp_faction_events.1.d
	flavor = mp_faction_events.1.f

	duration = 3

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by je_mp_faction_leader immediate only
	}

	immediate = {
		set_variable = { name = intl_strength_anarchist value = 20 }
		set_variable = { name = intl_strength_mutualist value = 20 }
		set_variable = { name = intl_strength_syndicalist value = 20 }
		set_variable = { name = intl_strength_blanquist value = 20 }
		set_variable = { name = intl_strength_marxist value = 20 }

		add_modifier = {
			name = mp_faction_extra_agitator_slots
			days = long_modifier_time
		}

		create_character = {
			template = ecchi_rus_bakunin_character_template
			on_created = {
				set_variable = bakunin_var
				save_scope_as = bakunin_scope
			}
		}
		scope:bakunin_scope = {
			add_character_flag = intl_delegate_anarchist
		}



		create_character = {
			template = intl_proudhon_character_template
			on_created = {
				set_variable = proudhon_var
				save_scope_as = proudhon_scope
			}
		}

		create_character = {
			template = ecchi_bel_de_paepe_character_template
			on_created = {
				set_variable = bel_de_paepe_var
				save_scope_as = bel_de_paepe_scope
			}
		}


		create_character = {
			template = fra_blanqui_character_template
			on_created = {
				set_variable = blanqui_var
				save_scope_as = blanqui_scope
			}
		}
		

		create_character = {
			template = agitator_karl_marx
			on_created = {
				set_variable = karl_marx_var
				save_scope_as = karl_marx_scope
			}
		}
	}

	option = {
		name = mp_faction_events.1.a
		default_option = yes
		set_variable = {
			name = mp_faction_supported_ig
			value = 1
		}
		set_variable = mp_faction_leader_confirmed
	}

	option = {
		name = mp_faction_events.1.b
		set_variable = {
			name = mp_faction_supported_ig
			value = 2
		}
		set_variable = mp_faction_leader_confirmed
	}

	option = {

			name = mp_faction_supported_ig
			value = 3
		}
		set_variable = mp_faction_leader_confirmed
	}

	option = {
		name = mp_faction_events.1.d
		set_variable = {
			name = mp_faction_supported_ig
			value = 4
		}
		set_variable = mp_faction_leader_confirmed
	}

	option = {
		name = mp_faction_events.1.e
		set_variable = {
			name = mp_faction_supported_ig
			value = 5
		}
		set_variable = mp_faction_leader_confirmed
